## Редица A

```py
def generateAn(n):
    if n < 0:
        return 0
    if n == 0:
        return 1
    if n == 1:
        return 2
    if n == 2:
        return 4
    if n == 3:
        return 8
    return generateAn(n - 1) * generateAn(n - 3) - generateAn(n - 2) * generateAn(n - 4)


def generateFirstN(n):
    lst = []
    for i in range(n):
        lst.append(generateAn(i))
    return lst
```
